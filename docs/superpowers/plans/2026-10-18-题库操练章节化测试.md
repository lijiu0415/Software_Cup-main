# 题库操练章节化测试 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将题库操练改为「按章节整章测试」：每章固定 5 单选 + 5 判断 + 3 多选 + 2 编程题（先做人工智能导论前 3 章），章节解锁后一次性作答统一提交判分，测试结果按规则更新画像并按「仅留最新 + 往期按正确率 + 易错累计」存储。

**Architecture:** 新增 `chapter_bank.py` 题库数据（45 题）+ 两张测试结果表（`chapter_test` 最新 / `chapter_test_history` 历史），在 `exercise.py` 增加章节/取题/提交/结果四个接口；复用 `AnswerRecord` 实现易错知识点累计；前端 `ExercisePage` 改为章节下拉 + 整章分组作答。

**Tech Stack:** FastAPI + SQLAlchemy(async) + Pydantic；React 18 + Ant Design 5 + Zustand。

## Global Constraints

- 项目**不是 git 仓库**，无 commit 步骤；每个任务以验证命令收尾。
- 后端语法验证用 `python -m py_compile`；前端构建验证用 `npm run build`。
- 画像枚举约束复用 `agents/diagnose_agent.py` 的 `ALLOWED_VALUES`；`error_type ∈ {概念混淆, 计算错误, 逻辑漏洞, 实践不足}`；`base_level/learn_speed` 为 1–5 整数。
- 章节解锁条件：该章所有 `KnowledgePoint` 的 `StudentMastery.completed == True`。
- 正确率分母 `total = 13`（5 单选 + 5 判断 + 3 多选，不含 2 道编程题）；编程题不判分、不写入 `AnswerRecord`。
- `get_db` 依赖会在端点正常返回后自动 `commit`，端点内**不要**手动 `db.commit()`。
- 章节命名 `{course前缀}_ch{NN}_{序号}`，如 `ai_ch01_01`；序号 01–05 单选、06–10 判断、11–13 多选、14–15 编程。
- 章节 `chapter_id` 前缀 `ai_` → 人工智能导论、`sc_` → 软件安全技术。

async def get_chapter_test(db: AsyncSession, student_id: str, chapter_id: str) -> Optional[ChapterTest]:
    """获取某生某章最新一次测试结果"""
    result = await db.execute(
        select(ChapterTest).where(and_(
            ChapterTest.student_id == student_id,
            ChapterTest.chapter_id == chapter_id,
        ))
    )
    return result.scalar_one_or_none()


async def upsert_chapter_test(
    db: AsyncSession,
    student_id: str,
    chapter_id: str,
    course: str,
    correct_count: int,
    wrong_count: int,
    total_count: int,
    accuracy: float,
    detail: str,
    profile_changes: str,
) -> ChapterTest:
    """写入或覆盖某生某章测试结果（仅保留最新）"""
    from datetime import datetime
    existing = await get_chapter_test(db, student_id, chapter_id)
    if existing:
        existing.course = course
        existing.correct_count = correct_count
        existing.wrong_count = wrong_count
        existing.total_count = total_count
        existing.accuracy = accuracy
        existing.detail = detail
        existing.profile_changes = profile_changes
        existing.create_time = datetime.now()
        await db.flush()
        return existing
    record = ChapterTest(
        student_id=student_id, chapter_id=chapter_id, course=course,
        correct_count=correct_count, wrong_count=wrong_count, total_count=total_count,
        accuracy=accuracy, detail=detail, profile_changes=profile_changes,
    )
    db.add(record)
    await db.flush()
    return record


async def add_chapter_test_history(
    db: AsyncSession,
    student_id: str,
    chapter_id: str,
    course: str,
    correct_count: int,
    wrong_count: int,
    total_count: int,
    accuracy: float,
) -> ChapterTestHistory:
    """追加一条章节测试历史（仅正确率，无 detail）"""
    record = ChapterTestHistory(
        student_id=student_id, chapter_id=chapter_id, course=course,
        correct_count=correct_count, wrong_count=wrong_count, total_count=total_count,
        accuracy=accuracy,
    )
    db.add(record)
    await db.flush()
    return record


async def get_chapter_test_history(db: AsyncSession, student_id: str, chapter_id: str) -> List[ChapterTestHistory]:
    """获取某生某章测试历史（新→旧）"""
    result = await db.execute(
        select(ChapterTestHistory)
        .where(and_(
            ChapterTestHistory.student_id == student_id,
            ChapterTestHistory.chapter_id == chapter_id,
        ))
        .order_by(ChapterTestHistory.create_time.desc())
    )
    return result.scalars().all()

def _chapter_course(chapter_id: str) -> str:
    """由 chapter_id 前缀映射课程名"""
    if chapter_id.startswith("ai_"):
        return "人工智能导论"
    if chapter_id.startswith("sc_"):
        return "软件安全技术"
    return ""


async def _chapter_progress(db: AsyncSession, student_id: str, chapter_id: str):
    """返回 (total_kp, completed_kp)；total=0 表示章节无知识点"""
    kps = await crud.get_kps_by_chapter(db, chapter_id)
    masteries = await crud.get_all_student_masteries(db, student_id)
    completed_ids = {m.kp_id for m in masteries if m.completed}
    total = len(kps)
    completed = sum(1 for kp in kps if kp.kp_id in completed_ids)
    return total, completed


def _serialize_question(q: dict) -> dict:
    """序列化题目：不含答案/解析；编程题额外给出参考解答（自评）"""
    item = {
        "ex_id": q["ex_id"],
        "kp_id": q.get("kp_id"),
        "type": q["type"],
        "difficulty": q.get("difficulty", 3),
        "question": q["question"],
        "options": q.get("options") or [],
    }
    if q["type"] == "编程题":
        item["reference_answer"] = q.get("reference_answer", "")
    return item

def _judge(question: dict, student_answer: Any):
    """判分：返回 is_correct；编程题返回 None（不判分）"""
    qtype = question["type"]
    if qtype == "编程题":
        return None
    if qtype == "多选题":
        a = sorted([int(x) for x in (question["answer"] or [])])
        b = sorted([int(x) for x in (student_answer or [])])
        return a == b
    return student_answer == question["answer"]


def _apply_test_rules(profile: dict, acc: float, wrong_by_type: dict) -> List[dict]:
    """根据测试结果返回画像变更列表 [{field, old, new}]"""
    changes = []
    base = int(profile.get("base_level") or 3)
    new_base = base
    if acc >= 80:
        new_base = min(5, base + 1)
    elif acc < 40:
        new_base = max(1, base - 1)
    if new_base != base:
        changes.append({"field": "base_level", "old": base, "new": new_base})

    wrong = sum(wrong_by_type.values())
    if wrong >= 2:
        ordered = sorted(wrong_by_type.items(), key=lambda x: -x[1])
        if ordered and ordered[0][1] > 0 and (len(ordered) < 2 or ordered[0][1] != ordered[1][1]):
            dominant = ordered[0][0]
            new_err = "逻辑漏洞" if dominant == "多选题" else "概念混淆"
            old_err = profile.get("error_type")
            if new_err in ALLOWED_VALUES["error_type"] and new_err != old_err:
                changes.append({"field": "error_type", "old": old_err, "new": new_err})
    return changes
```

@router.post("/submit_test")
async def submit_chapter_test(
    req: SubmitTestRequest,
    db: AsyncSession = Depends(get_db),
    current_user: AuthUser = Depends(get_current_user),
):
    """提交整章测试：判分 + 记录 + 更新画像 + 存储结果"""
    student_id = resolve_student_id(req.student_id, current_user)
    chapter_id = req.chapter_id

    total, completed = await _chapter_progress(db, student_id, chapter_id)
    if total == 0 or completed < total:
        return JSONResponse(status_code=403, content={"error": "该章节尚未学完全部知识点，暂无法测试"})

    questions = [q for q in CHAPTER_BANK if q["chapter_id"] == chapter_id]
    if not questions:
        return {"error": "该章节暂无题目"}

    answer_map = {a.get("ex_id"): a.get("student_answer") for a in (req.answers or [])}

    correct = 0
    wrong = 0
    wrong_by_type = {"单选题": 0, "判断题": 0, "多选题": 0}
    detail = []

    for q in questions:
        qtype = q["type"]
        student_answer = answer_map.get(q["ex_id"])
        is_correct = _judge(q, student_answer)

        if is_correct is None:  # 编程题不判分、不记录
            continue

        if is_correct:
            correct += 1
        else:
            wrong += 1
            wrong_by_type[qtype] = wrong_by_type.get(qtype, 0) + 1

        correct_text = _format_answer(q, q["answer"])
        student_text = _format_answer(q, student_answer)
        detail.append({
            "ex_id": q["ex_id"],
            "type": qtype,
            "kp_id": q.get("kp_id"),
            "is_correct": is_correct,
            "student_answer": student_text,
            "correct_answer": correct_text,
            "analysis": q.get("analysis", ""),
        })

        # 全部判分题写入答题记录（对+错），用于错题本/易错知识点累计
        await crud.create_answer_record(
            db,
            student_id=student_id,
            kp_id=q.get("kp_id"),
            ex_id=q["ex_id"],
            type=qtype,
            question=q["question"],
            options=json.dumps(q.get("options") or [], ensure_ascii=False),
            student_answer=student_text,
            correct_answer=correct_text,
            is_correct=is_correct,
            analysis=q.get("analysis", ""),
        )

    total_count = correct + wrong
    acc = round(correct / total_count * 100, 1) if total_count else 0.0

    # 画像更新
    student = await crud.get_student(db, student_id)
    profile = {
        "base_level": student.base_level if student else 3,
        "error_type": student.error_type if student else "概念混淆",
    }
    changes = _apply_test_rules(profile, acc, wrong_by_type)
    if changes:
        await crud.update_student(db, student_id, **{c["field"]: c["new"] for c in changes})

    course = student.course if student and student.course else "人工智能导论"
    detail_json = json.dumps(detail, ensure_ascii=False)
    changes_json = json.dumps(changes, ensure_ascii=False)

    await crud.upsert_chapter_test(
        db, student_id, chapter_id, course,
        correct_count=correct, wrong_count=wrong, total_count=total_count,
        accuracy=acc, detail=detail_json, profile_changes=changes_json,
    )
    await crud.add_chapter_test_history(
        db, student_id, chapter_id, course,
        correct_count=correct, wrong_count=wrong, total_count=total_count, accuracy=acc,
    )

    return {
        "accuracy": acc,
        "correct": correct,
        "wrong": wrong,
        "total": total_count,
        "profile_changes": changes,
        "detail": detail,
    }

      {/* 右侧 数据栏 */}
      <div style={{ flex: '0 0 22%', display: 'flex', flexDirection: 'column', gap: 12 }}>
        <Card className="glass-card" size="small" title="练习数据">
          <Statistic title="已做题数" value={stats.done} suffix="题" />
          <Statistic title="正确率" value={correctRate} suffix="%" style={{ marginTop: 12 }} />
          <Statistic title="错题数" value={stats.wrong} suffix="题" style={{ marginTop: 12 }} />
        </Card>
        <Card className="glass-card" size="small" title="易错知识点排行">
          {weakPoints.length === 0 ? (
            <Empty description="暂无错题数据" image={Empty.PRESENTED_IMAGE_SIMPLE} />
          ) : (
            <List
              size="small"
              dataSource={weakPoints}
              renderItem={(p, i) => (
                <List.Item style={{ border: 'none', padding: '6px 0' }}>
                  <Space>
                    <Tag color={i === 0 ? 'red' : i === 1 ? 'orange' : 'blue'}>{i + 1}</Tag>
                    <Text style={{ fontSize: 13 }}>{p.name}</Text>
                    <Text type="secondary" style={{ fontSize: 12 }}>错 {p.wrong_count} 次</Text>
                  </Space>
                </List.Item>
              )}
            />
          )}
        </Card>
      </div>
    </div>
  )
}
```
